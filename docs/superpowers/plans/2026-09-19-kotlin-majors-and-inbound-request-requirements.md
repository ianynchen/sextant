```java
    private static List<Boolean> declaredDefaults(List<KmValueParameter> parameters) {

    }
```

```python
import re
from dataclasses import dataclass

from sextant.domain.symbols import bare_descriptor  # the helper every engine uses; check its module path with grep

_DEFAULT_SUFFIX = "$default"
_MARKER = "Ljava/lang/Object;"
_DESCRIPTOR = re.compile(r"^\((?P<args>.*)\)(?P<ret>.+)$")


@dataclass(frozen=True)
class _Bridge:
    """One `$default` bridge the facts declared, joined to its sibling (spec §3a). `receiver`
    is 1 where the sibling is an instance method and the bridge's slot 0 is its receiver, 0
    for a static sibling; `declared` is the sibling's own slot count."""

    owner: str
    name: str
    descriptor: str
    sibling_descriptor: str
    receiver: int
    declared: int
    sibling_static: bool


def _slots(descriptor: str) -> tuple[list[str], str]:
    """The argument descriptors and the return, split as the JVM does (JVMS 4.3.3)."""
    match = _DESCRIPTOR.match(descriptor)
    if match is None:
        return [], descriptor
    args: list[str] = []
    text = match.group("args")
    i = 0
    while i < len(text):
        start = i
        while text[i] == "[":
            i += 1
        if text[i] == "L":
            i = text.index(";", i) + 1
        else:
            i += 1
        args.append(text[start:i])
    return args, match.group("ret")


def _masks(parameters: int) -> int:
    return (parameters + 31) // 32


def _bridges(workspace: Workspace) -> dict[tuple[str, str, str], _Bridge]:
    """Keyed on (owner, name, descriptor) of the BRIDGE. A bridge whose sibling cannot be
    settled — neither candidate declared, or both (A5) — is not in the map, so its calls
    stay exactly as the bytecode said."""
    declared: dict[tuple[str, str], tuple[str, bool]] = {}   # (owner, name+descriptor) → (symbol, static)
    for row in workspace.query("method_declarations_with_static"):   # see Step 3b
        ...
    found: dict[tuple[str, str, str], _Bridge] = {}
    for row in workspace.query("bridge_declarations"):
        symbol = bare_descriptor(str(row["symbol"]))
        owner, rest = symbol.split("#", 1)
        name = rest.split("(", 1)[0]
        if not name.endswith(_DEFAULT_SUFFIX):
            continue   # a DefaultImpls body also carries bridge_of; dispatch.py owns those
        descriptor = "(" + rest.split("(", 1)[1]
        args, ret = _slots(descriptor)
        if not args or args[-1] != _MARKER:
            continue
        function = str(row["bridge_of"])
        candidates: list[_Bridge] = []
        for receiver in (1, 0):
            available = len(args) - receiver - 1
            for n in range(1, available + 1):
                if n + _masks(n) != available:
                    continue
                if any(a != "I" for a in args[receiver + n : len(args) - 1]):
                    break
                sibling = "(" + "".join(args[receiver : receiver + n]) + ")" + ret
                hit = declared.get((owner + "#", function + sibling))
                if hit is not None:
                    candidates.append(_Bridge(owner + "#", name, descriptor, sibling, receiver, n, hit[1]))
                break
        if len(candidates) == 1:
            found[(owner + "#", name, descriptor)] = candidates[0]
    return found
```

```python
def _effective(workspace: Workspace) -> list[dict[str, Any]]:
    bridges = _bridges(workspace)
    rows = []
    for site in workspace.query("call_sites"):
        key = (str(site["owner"]), str(site["name"]), str(site["descriptor"]))
        bridge = bridges.get(key)
        rows.append(site if bridge is None else _through_bridge(site, bridge) or site)
    return rows


def _through_bridge(site: dict[str, Any], bridge: _Bridge) -> dict[str, Any] | None:
    """The call as the sibling sees it, or None where a mask is not an int literal — a
    wrong trimming is worse than the silence there is today (spec §3a)."""
    arguments = list(site["arguments"])
    expected = bridge.receiver + bridge.declared + _masks(bridge.declared) + 1
    if len(arguments) != expected:
        return None
    masks: list[int] = []
    for argument in arguments[bridge.receiver + bridge.declared : -1]:
        if argument.get("kind") != "int":
            return None
        masks.append(int(str(argument["value"])))
    projected: list[dict[str, Any]] = []
    for i in range(bridge.declared):
        bit = (masks[i // 32] >> (i % 32)) & 1
        projected.append(
            {"kind": "unresolved", "reason": "defaulted"} if bit else arguments[bridge.receiver + i]
        )
    row = dict(site)
    row["name"] = bridge.name[: -len(_DEFAULT_SUFFIX)]
    row["descriptor"] = bridge.sibling_descriptor
    row["arguments"] = projected
    row["evidence_bridge"] = f"{bridge.owner}{bridge.name}{bridge.descriptor}"
    if bridge.receiver:
        receiver = arguments[0]
        row["invoke_kind"] = "virtual"
        row["receiver"] = receiver.get("source")
        row["receiver_parameter"] = receiver.get("value") if receiver.get("kind") == "parameter" else None
    return row
```

`receiver`/`receiver_parameter` semantics: D00 §5 — `receiver` is the value id the receiver came from, `receiver_parameter` the caller's parameter index when it is one. Set both from the receiver-slot argument exactly as above; a receiver argument with neither (`field`, `unresolved`) leaves both NULL, which every origin check already treats as "not the origin".

```python
def _helper_routes(site: dict[str, Any], scope: _Scope) -> list[_Route]:
    """Spec 2026-09-19 §4a: a module-local call reached inside a routing body is followed
    when the body's origin flows into it — as the receiver or as an argument — and, for a
    body whose linking rule declared `follow_calls: all` (Javalin's thread-local builder),
    whenever the callee has a body at all. Inside the callee the parameter that received
    the origin is the new origin. Guards only: a visited set keyed on (callee, origin) and a
    depth bound of 16; exhausting the bound files one `routing_depth` question and NO row
    (A2 — a row would claim an address nobody read)."""
    context = scope.context
    callee = context.method_symbols.get(_bare_target(site))
    if callee is None or bare_descriptor(callee) not in context.by_bare_caller:
        return []
    origin: int | None = None
    if scope.origin_parameter is not None:
        if site["receiver"] is None and site.get("receiver_parameter") == scope.origin_parameter:
            origin = 0   # a member/extension receiver is the callee's slot 0
        else:
            for i, argument in enumerate(site["arguments"]):
                if argument.get("kind") == "parameter" and argument.get("value") == scope.origin_parameter:
                    origin = i
                    break
        if origin is None:
            return []
    elif not scope.follow_all:
        return []
    key = (bare_descriptor(callee), origin)
    if key in scope.visited:
        return []
    if scope.depth >= _HELPER_DEPTH:
        scope.questions.append({"question": "routing_depth", "candidates": json.dumps([callee]), ...})
        return []
    body = _scope_of(context.by_bare_caller[bare_descriptor(callee)], context)
    body = replace(body, origin_parameter=origin, follow_all=scope.follow_all,
                   visited=scope.visited | {key}, depth=scope.depth + 1, questions=scope.questions)
    routes = []
    for inner in body.sites:
        if inner["result"] is not None and inner["result"] in body.consumed:
            continue
        routes.extend(_routes_of(inner, body))
        if not _is_catalog_word(inner, context):
            routes.extend(_helper_routes(inner, body))
    return routes
```
